function withTimeout(promise, ms) {
  const timeout = new Promise((_, reject) => {
    setTimeout(() => {
      reject(new Error(`Operation timed out after ${ms}ms`));
    }, ms);
  });

  return Promise.race([promise, timeout]);
}

async function demoTimeout() {
  console.log("--- Timeout Pattern ---");

  try {
    const result = await withTimeout(
      simulateAsync("Fast Read", 1000),
      2000
    );
    console.log("Case 1 (fast):", result);
  } catch (error) {
    console.log("Case 1 error:", error.message);
  }

  try {
    const result = await withTimeout(
      simulateAsync("Slow Read", 5000),
      2000
    );
    console.log("Case 2 (slow):", result);
  } catch (error) {
    console.log("Case 2 error:", error.message);
  }
}
