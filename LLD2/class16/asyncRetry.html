Retry with async/await

Move manually through failure, waiting, a fresh promise, and eventual success.

Exact JavaScript code

async function retry(fn, retries = 3, delay = 1000) {
  for (let attempt = 1; attempt <= retries; attempt++) {
    try {
      console.log(`  Attempt ${attempt}...`);
      const result = await fn();
      console.log(`  Succeeded on attempt ${attempt}`);
      return result;
    } catch (error) {
      console.log(`  Attempt ${attempt} failed: ${error.message}`);
      if (attempt === retries) {
        throw new Error(
          `Failed after ${retries} attempts. Last error: ${error.message}`
        );
      }
      console.log(`  Waiting ${delay}ms before retry...`);
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
}

async function demoRetry() {
  console.log("--- Retry Pattern ---");

  const flakyOp = createFlakyOperation();

  try {
    const result = await retry(flakyOp, 5, 500);
    console.log("Final result:", result);
  } catch (error) {
    console.log("Gave up:", error.message);
  }
}
